# Create dns_nvue role
```
ubuntu@oob-mgmt-server:~/ansible_lab$ cat roles/dns_nvue/tasks/main.yml 
---
- name: Configure DNS servers with NVUE
  nvidia.nvue.command:
    template: "{{ lookup('template', 'dns_commands.j2') }}"
    apply: true
    assume_yes: true
ubuntu@oob-mgmt-server:~/ansible_lab$ 
```
excution order:
```
lookup('template', 'dns_commands.j2')
                ↓
Render Jinja2 on Ansible controller 
                ↓
Generate set/unset NVUE commands
                ↓
nvidia.nvue.command send to switch
                ↓
apply: true Excute nv config apply
```

# Create validation playbook, check syntex and run

ubuntu@oob-mgmt-server:~/ansible_lab$ cat playbooks/validate_dns.yml 
---
- name: Validate DNS configuration on spine switches
  hosts: spine_sw
  become: true
  gather_facts: false

  tasks:
    - name: Read NVUE DNS configuration
      ansible.builtin.command:
        cmd: nv show system dns server
      register: nvue_dns
      changed_when: false

    - name: Read generated resolver configuration
      ansible.builtin.command:
        cmd: cat /etc/resolv.conf
      register: resolv_conf
      changed_when: false

    - name: Validate desired DNS configuration
      ansible.builtin.assert:
        that:
          - "'10.0.0.100' in nvue_dns.stdout"
          - "'10.0.0.101' in nvue_dns.stdout"
          - "'1.1.1.1' not in nvue_dns.stdout"
          - "'8.8.8.8' not in nvue_dns.stdout"
          - "'nameserver 10.0.0.100' in resolv_conf.stdout"
          - "'nameserver 10.0.0.101' in resolv_conf.stdout"
          - "'nameserver 1.1.1.1' not in resolv_conf.stdout"
          - "'nameserver 8.8.8.8' not in resolv_conf.stdout"
        success_msg: >-
          DNS validation passed on {{ inventory_hostname }}:
          10.0.0.100 and 10.0.0.101 are configured.
        fail_msg: >-
          DNS validation failed on {{ inventory_hostname }}.
          NVUE output: {{ nvue_dns.stdout }}
          resolv.conf: {{ resolv_conf.stdout }}

- name: Roll back DNS servers on spine switches
